  \begin{center}

  \end{center}

  \begin{itemize}
    \item an ``image'' is not to be found \emph{within} the scientific image

  \hfill{\footnotesize\color{faint} cf.\ Putnam on Williams's ``absolute conception''}

  \begin{center}
    not a picture \emph{inside} us\\
    but a \emph{mode of approach} --- one that removes the subject
  \end{center}
\end{frame}

\section{The Danish story}

  \begin{center}

    {\small\color{faint} the System with no remainder, no open place}
  \end{center}
\end{frame}

\begin{frame}{Sibbern \& Møller}
  \begin{itemize}
    \item not adequate to the natural sciences
    \item not adequate to the richness of life
  \end{itemize}
  \vspace{1em}
  \begin{center}
    \alert{$\gamma\nu\tilde\omega\theta\iota\ \sigma\varepsilon\alpha\upsilon\tau\acute o\nu$}\\[4pt]
    {\small man is still \emph{becoming} --- not yet a god}
  \end{center}
\end{frame}

\begin{frame}{Kierkegaard}
  \begin{center}
    The speculative philosopher is \alert{forgetful.}
  \end{center}
  \vspace{1em}
  \begin{center}
    {\small what he has forgotten is \emph{who he himself is}}
  \end{center}
\end{frame}

\begin{frame}{Rasmus Nielsen \;(1809--1884)}
  \begin{center}
    Kierkegaard's mantle $\rightarrow$ philosophy of science\\[1em]
    \alert{objektiveringsloven}\\[4pt]
    {\small\color{faint} the law of objectification}
  \end{center}
\end{frame}

\begin{frame}{Harald Høffding}
  \begin{itemize}
    \item no objectification is ever \emph{final}
    \item science cannot be purged of its \emph{anthropocentric} elements
  \end{itemize}
  \vspace{1em}
  \begin{center}
    {\small\itshape ``When will the last anthropomorphism be found?''}
  \end{center}
\end{frame}

\begin{frame}{Niels Bohr}
  \begin{center}
    \large complementarity\\[8pt]
    {\small = Høffding's relationist epistemology,\\ \emph{physically implemented}}\\[10pt]
    \alert{the subject stays in the picture --- by necessity}
  \end{center}
\end{frame}
